Rg                                                                          *rg*

SEARCH BY FILETYPE
>bash
    rg xxx -t lua -t js
<
Exclude a certain type ~
>bash
    rg xxx -T lua -T js
<

SEARCH DIR
>bash
    rg xxx src/
<

GLOB
>bash
    rg xxx -g "src/**" -g "lua/**"
    rg xxx -g "!node_modules/**"
>

REGEX

>bash
    rg '^hello'                 Match at beginning of line
    rg 'hello$'                 Match at end of line
    rg '\bhello\b'              Match whole word
    rg 'foo|bar'                Match foo OR bar
    rg 'foo.*bar'               Match foo followed by bar on the same line
    rg 'foo\s+bar'              Match whitespace between foo and bar
    rg 'foo.{0,10}bar'          Match foo and bar within 10 characters
>

OTHER

show file name ~
>bash
    rg hello -l
>

match word ~
>bash
    rg hello -w
>

match multiline ~
>bash
    rg "hello\nworld" -U
>

case sensitive ~
>bash
    rg hello -s
>

no ignore ~
>bash
    rg hello --no-ignore
>

FZF                                                                        *fzf*
>txt
    foo                         Fuzzy-match
    'foo                        Exact match
    'wild'                      Items that include wild at word boundaries
    ^foo                        Prefix match
    foo$                        Suffix match
    !foo                        Exclude items matching foo
    !'foo                       Exclude items containing exact substring foo
    !^foo                       Exclude items starting with foo
    !foo$                       Exclude items ending with foo
    foo | bar                   OR: match foo or bar
    !foo bar                    Match bar, but exclude foo
    ^src/ *.lua                 Starting with src/ and fuzzy pattern *.lua
>
